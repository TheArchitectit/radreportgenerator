using System.IO;
using System.Threading;
using System.Threading.Tasks;
using OpenReportViewer.Core.Interfaces;
using OpenReportViewer.Core.Models;

namespace OpenReportViewer.Reporting
{
    public interface IPdfReportGenerator
    {
        Task<Stream> GenerateAsync(IReportData data, PdfGenerationOptions options, CancellationToken cancellationToken = default);
        ValidationResult Validate(IReportData data);
    }

    public record PdfGenerationOptions
    {
        public bool IncludeCoverPage { get; init; } = true;
        public bool IncludeExecutiveSummary { get; init; } = true;
        public bool IncludeVmInventory { get; init; } = true;
        public bool IncludeCharts { get; init; } = true;
        public string FontFamily { get; init; } = "Segoe UI";
    }

    public record ValidationResult
    {
        public bool IsValid { get; init; }
        public System.Collections.Generic.List<string> Errors { get; init; } = new();
        public System.Collections.Generic.List<string> Warnings { get; init; } = new();
    }
}

using QuestPDF.Fluent;
using QuestPDF.Helpers;
using QuestPDF.Infrastructure;
using Microsoft.Extensions.Logging;
using OpenReportViewer.Core.Interfaces;
using OpenReportViewer.Core.Models;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

namespace OpenReportViewer.Reporting
{
    public class QuestPdfReportGenerator : IPdfReportGenerator
    {
        private readonly ILogger<QuestPdfReportGenerator> _logger;
        private readonly IChartProvider _chartProvider;
        
        public QuestPdfReportGenerator(ILogger<QuestPdfReportGenerator> logger, IChartProvider chartProvider)
        {
            _logger = logger ?? throw new ArgumentNullException(nameof(logger));
            _chartProvider = chartProvider ?? throw new ArgumentNullException(nameof(chartProvider));
            QuestPDF.Settings.License = LicenseType.Community;
        }
        
        public async Task<Stream> GenerateAsync(IReportData data, PdfGenerationOptions options, CancellationToken cancellationToken = default)
        {
            _logger.LogInformation("Starting PDF generation for {VmCount} VMs", data.VirtualMachines.Count);
            var validation = Validate(data);
            if (!validation.IsValid)
            {
                throw new InvalidOperationException($"Validation failed: {string.Join(", ", validation.Errors)}");
            }
            var document = Document.Create(container =>
            {
                container.Page(page =>
                {
                    page.Size(PageSizes.A4);
                    page.Margin(2, Unit.Centimetre);
                    page.DefaultTextStyle(x => x.FontFamily(options.FontFamily).FontSize(11));
                    page.Header().Element(ComposeHeader);
                    page.Content().Element(content => ComposeContent(content, data, options));
                    page.Footer().Element(ComposeFooter);
                });
            });
            var stream = new MemoryStream();
            document.GeneratePdf(stream);
            stream.Position = 0;
            _logger.LogInformation("PDF complete: {Pages} pages, {KB:F1} KB", document.PageCount, stream.Length / 1024.0);
            return stream;
        }
        
        public ValidationResult Validate(IReportData data)
        {
            var result = new ValidationResult();
            if (data == null) result.Errors.Add("Data cannot be null");
            if (data?.VirtualMachines == null) result.Warnings.Add("No VMs in dataset");
            result.IsValid = !result.Errors.Any();
            return result;
        }
        
        private void ComposeHeader(IContainer container)
        {
            container.Row(row =>
            {
                row.RelativeItem().Text("OpenReportViewer").Bold().FontSize(14);
                row.ConstantItem(100).AlignRight().Text(DateTime.Now.ToString("MMM dd, yyyy"));
            });
        }
        
        private void ComposeContent(IContainer container, IReportData data, PdfGenerationOptions options)
        {
            container.Column(column =>
            {
                column.Item().Element(ComposeCoverPage);
                column.Item().PageBreak();
                column.Item().Element(ComposeExecutiveSummary);
            });
        }
        
        private void ComposeCoverPage(IContainer container)
        {
            container.PaddingTop(8, Unit.Centimetre).AlignCenter().Column(column =>
            {
                column.Item().Text("Infrastructure Assessment Report").FontSize(24).Bold().FontColor(Colors.Blue.Darken2);
                column.Item().PaddingTop(2, Unit.Centimetre).Text($"Generated: {DateTime.Now:MMMM dd, yyyy}");
                column.Item().Text($"VMs Analyzed: {data.VirtualMachines.Count:N0}");
            });
        }
        
        private void ComposeExecutiveSummary(IContainer container)
        {
            container.Column(column =>
            {
                column.Item().Text("Executive Summary").FontSize(16).Bold();
                column.Item().PaddingTop(10).Text("This report provides infrastructure analysis based on provided data export.");
            });
        }
        
        private void ComposeFooter(IContainer container)
        {
            container.AlignCenter().Text(x =>
            {
                x.Span("Page ");
                x.CurrentPageNumber();\n                x.Span(" of ");
                x.TotalPages();
            });
        }
    }
